import java.util.*;

class Wrestler implements Comparable<Wrestler>
{
  private String wname;
  private int weight;

  Wrestler(String wname, int weight)  
    {
      this.wname=wname;
      this.weight=weight;
    }

  public String getWname()
   {
     return wname;
   }

  public int getWeight()
   {
     return weight;
   }

  public String toString()
   {
     return "(" + wname + "=" + weight + ")";
   }

  public int compareTo(Wrestler obj)
   {
     //return this.weight - obj.getWeight();
     return this.wname.compareTo(obj.getWname());
   }
}

class Demo
{
  public static void main(String args[ ])
   {
      ArrayList<Wrestler> list=new ArrayList<Wrestler>();

      list.add(new Wrestler("Vasant", 120));
      list.add(new Wrestler("Mahesh", 125));
      list.add(new Wrestler("Zaheer", 121));
      list.add(new Wrestler("Anil", 100));
      list.add(new Wrestler("Ramesh", 90));

      System.out.println(list);
      Collections.sort(list);
      System.out.println(list);
   }
}
